;; FMD SEIRC-V AGENT-BASED MODEL  --  BARRIER VACCINATION

;; strategy-specific block (domain geometry and vaccination rules) differs.

;;##########################################################################
;; COMMON ENGINE v2  --  IDENTICAL IN ALL FIVE STRATEGY FILES
;; (prophylactic_HIT, prophylactic_CONTROL, ring, barrier, spatial_CONTROL)
;;
;; Everything between the "BEGIN COMMON ENGINE" and "END COMMON ENGINE" markers
;; is byte-for-byte the same in every file; the files differ only in the
;; "STRATEGY-SPECIFIC BLOCK" below the engine (domain, vaccination rules).
;;##########################################################################
;; ===== BEGIN COMMON ENGINE =====

globals [
  ;; scenario (set by set-scenario / set-disease-parameters)
  current-serotype Ro VP
  exposure-period subclinical-period clinical-period carrier-duration
  carrier-probability n-inf-days

  ;; transmission
  beta-daily           ;; per-contact daily infection probability (calibrated so lifetime R0 = Ro)
  calibrated-R0        ;; expected secondary infections per case at the calibrated beta
  sample-klists        ;; kernel values of a random sample of animals (calibration)

  ;; domain (set by set-domain in the strategy-specific block)
  r3 area-km2 density population simulation-days

  ;; vaccination bookkeeping
  HIT                  ;; 1 - 1/Ro                                           Eq. (1)
  vaccination-coverage ;; proportion of eligible animals vaccinated (prophylactic: V* = HIT/VP, Eq. 2)
  protected-count      ;; vaccinated animals that became protected (V) through vaccination
  doses-given          ;; animals vaccinated (responders + non-responders)

  ;; compartment counts
  S-count E-count I-sub-count I-clin-count C-count R-count V-count

  ;; outputs
  cumulative-infections    ;; S -> E events after the index case
  peak-clinical peak-day epidemic-duration
  attack-rate              ;; (cumulative-infections + 1) / population
  major-outbreak           ;; 1 if attack-rate > 0.05
  balance-error            ;; population - sum of compartments (must be 0)

]

turtles-own [
  state                ;; S, E, I_sub, I_clin, C, R, V
  days-in-state
  initial-source?
  vaccinated?          ;; has received a dose (responder or not)
  scheduled?           ;; selected for the reactive campaign (ring/barrier)
  nbr-list             ;; the n-contacts nearest animals (potential contacts)
  nbr-k                ;; kernel K(d) = exp(-alpha d) for each contact
]

;;--------------------------------------------------------------------------
;; SETUP
;;--------------------------------------------------------------------------
to setup
  clear-all
  set-scenario
  set-disease-parameters
  set simulation-days 350
  set cumulative-infections 0
  set peak-clinical 0
  set peak-day 0
  set epidemic-duration 0
  set protected-count 0
  set doses-given 0

  set-domain                                   ;; strategy-specific (population, area)
  create-turtles population [ initialise-animal ]
  place-animals                                ;; strategy-specific (geometry)
  paint-domain                                 ;; strategy-specific (display only)

  build-contacts
  calibrate-beta                               ;; herd is fully susceptible here
  strategy-setup                               ;; prophylactic vaccination happens here (t = 0)
  introduce-index

  update-compartment-counts
  reset-ticks
  update-outputs
  log-setup
end

to initialise-animal
  set size 0.3
  set shape "circle"
  set state "S"
  set color brown
  set days-in-state 0
  set initial-source? false
  set vaccinated? false
  set scheduled? false
  set nbr-list []
  set nbr-k []
end

to set-scenario
  if scenario-choice = "O_R2"     [ set current-serotype "O"     set Ro 2 ]
  if scenario-choice = "O_R4"     [ set current-serotype "O"     set Ro 4 ]
  if scenario-choice = "A_R2"     [ set current-serotype "A"     set Ro 2 ]
  if scenario-choice = "A_R4"     [ set current-serotype "A"     set Ro 4 ]
  if scenario-choice = "Asia1_R2" [ set current-serotype "Asia1" set Ro 2 ]
  if scenario-choice = "Asia1_R4" [ set current-serotype "Asia1" set Ro 4 ]
end

;; durations (days) and vaccine protectiveness per serotype
to set-disease-parameters
  if current-serotype = "O" [
    set exposure-period 1.2  set subclinical-period 1.7  set clinical-period 6.6  set VP 0.823
  ]
  if current-serotype = "A" [
    set exposure-period 1.0  set subclinical-period 2.8  set clinical-period 7.5  set VP 0.767
  ]
  if current-serotype = "Asia1" [
    set exposure-period 1.5  set subclinical-period 2.0  set clinical-period 7.0  set VP 0.787
  ]
  set carrier-duration 320
  set carrier-probability 0.5
  ;; integer number of days an animal transmits in the acute phase: progression is tested as
  ;; days-in-state >= duration on an integer counter, so 1.7 d -> 2 d, 6.6 d -> 7 d, etc.
  set n-inf-days (ceiling subclinical-period) + (ceiling clinical-period)
end

;;--------------------------------------------------------------------------
;; CONTACT STRUCTURE AND TRANSMISSION CALIBRATION
;;--------------------------------------------------------------------------
;; Each animal has n-contacts potential contacts: its nearest animals (any state).
;; Daily infection probability from infectious i to susceptible j is beta-daily * K(d_ij),
;; K(d) = exp(-alpha d)  (d in km; Eq. 4).
to build-contacts
  if n-contacts >= population [
    user-message "n-contacts must be smaller than the population."
    stop
  ]
  let search-r 1.5 * (sqrt (n-contacts / (pi * density)))
  ask turtles [
    let r search-r
    let cand other turtles in-radius r
    while [ count cand < n-contacts ] [
      set r r * 1.5
      set cand other turtles in-radius r
    ]
    let nearest min-n-of n-contacts cand [ distance myself ]
    set nbr-list sort-on [ distance myself ] nearest
    set nbr-k map [ nb -> exp (0 - (alpha * (distance nb))) ] nbr-list
  ]
end

;; probability that a given contact j is infected at some time during the whole infectious
;; life of i (acute phase, plus the carrier phase if i becomes a carrier)
to-report pair-infection-probability [ b k ]
  let q min (list 1 (b * k))
  let p-acute 1 - ((1 - q) ^ n-inf-days)
  let p-carrier 1 - ((1 - (min (list 1 (carrier-infectiousness * q)))) ^ carrier-duration)
  report 1 - ((1 - p-acute) * (1 - (carrier-probability * p-carrier)))
end

to-report expected-secondary [ b ]
  report mean map [ ks -> sum map [ k -> pair-infection-probability b k ] ks ] sample-klists
end

;; Solve for beta-daily so that the expected number of secondary infections caused by one
;; infected animal in a fully susceptible herd equals Ro (bisection).
to calibrate-beta
  let sample-size min (list 300 (count turtles))
  set sample-klists [ nbr-k ] of (n-of sample-size turtles)
  ifelse (expected-secondary 1) < Ro [
    set beta-daily 1
    show (word "WARNING: Ro = " Ro " cannot be reached with n-contacts = " n-contacts "; beta-daily set to 1.")
  ] [
    let lo 0
    let hi 1
    repeat 50 [
      let mid (lo + hi) / 2
      ifelse (expected-secondary mid) < Ro [ set lo mid ] [ set hi mid ]
    ]
    set beta-daily (lo + hi) / 2
  ]
  set calibrated-R0 expected-secondary beta-daily
end

;;--------------------------------------------------------------------------
;; INDEX CASE, VACCINATION, STATE CHANGE
;;--------------------------------------------------------------------------
;; the clinically infectious index animal is the susceptible animal nearest the domain centre
to introduce-index
  let src min-one-of (turtles with [ state = "S" ]) [ distancexy 0 0 ]
  if src != nobody [
    ask src [
      set-state "I_clin"
      set initial-source? true
      set size 0.8
    ]
  ]
end

to set-state [ new-state ]
  set state new-state
  set days-in-state 0
  if new-state = "S"      [ set color brown ]
  if new-state = "E"      [ set color yellow ]
  if new-state = "I_sub"  [ set color orange ]
  if new-state = "I_clin" [ set color red ]
  if new-state = "C"      [ set color gray ]
  if new-state = "R"      [ set color cyan ]
  if new-state = "V"      [ set color green ]
end

;; one dose: the animal becomes protected (V) with probability VP, otherwise it stays
;; susceptible (vaccine failure). Only susceptible animals are ever vaccinated.
to vaccinate
  set vaccinated? true
  set doses-given doses-given + 1
  if random-float 1 < VP [
    set-state "V"
    set protected-count protected-count + 1
  ]
end

to become-exposed
  set-state "E"
  set cumulative-infections cumulative-infections + 1
end

;;--------------------------------------------------------------------------
;; GO  (one tick = one day)
;;   1 strategy-daily (reactive vaccination)  2 progression  3 transmission  4 outputs
;;--------------------------------------------------------------------------
to go
  if ticks >= simulation-days [ stop ]
  strategy-daily
  update-states
  transmit
  update-compartment-counts
  update-outputs
  tick
end

;; time-based progression on an integer day counter (fractional durations act as the next
;; whole day); an animal that changes state on a tick does not change again on that tick
to update-states
  ask turtles [
    set days-in-state days-in-state + 1
    if state = "V" and days-in-state >= immunity-duration [
      set-state "S"
    ]
    if state = "E" and days-in-state >= exposure-period [
      set-state "I_sub"
    ]
    if state = "I_sub" and days-in-state >= subclinical-period [
      set-state "I_clin"
    ]
    if state = "I_clin" and days-in-state >= clinical-period [
      ifelse random-float 1 < carrier-probability [ set-state "C" ] [ set-state "R" ]
    ]
    if state = "C" and days-in-state >= carrier-duration [
      set-state "R"
    ]
  ]
end

;; subclinical and clinical animals infect each susceptible contact with probability
;; beta-daily * K(d); carriers do so with carrier-infectiousness * beta-daily * K(d).
;; Protected (V), recovered and already-infected animals cannot be infected.
to transmit
  ask turtles with [ state = "I_sub" or state = "I_clin" or state = "C" ] [
    let rel ifelse-value (state = "C") [ carrier-infectiousness ] [ 1 ]
    if rel > 0 [
      (foreach nbr-list nbr-k [ [ nb k ] ->
        if [ state ] of nb = "S" [
          if random-float 1 < (rel * beta-daily * k) [
            ask nb [ become-exposed ]
          ]
        ]
      ])
    ]
  ]
end

;;--------------------------------------------------------------------------
;; OUTPUTS AND CHECKS
;;--------------------------------------------------------------------------
to update-compartment-counts
  set S-count      count turtles with [ state = "S" ]
  set E-count      count turtles with [ state = "E" ]
  set I-sub-count  count turtles with [ state = "I_sub" ]
  set I-clin-count count turtles with [ state = "I_clin" ]
  set C-count      count turtles with [ state = "C" ]
  set R-count      count turtles with [ state = "R" ]
  set V-count      count turtles with [ state = "V" ]
end

to update-outputs
  if I-clin-count > peak-clinical [
    set peak-clinical I-clin-count
    set peak-day ticks
  ]
  if (E-count + I-sub-count + I-clin-count) > 0 [ set epidemic-duration ticks ]
  set attack-rate (cumulative-infections + 1) / population
  set major-outbreak ifelse-value (attack-rate > 0.05) [ 1 ] [ 0 ]
  set balance-error population - (S-count + E-count + I-sub-count + I-clin-count + C-count + R-count + V-count)
end

to-report protected-fraction
  report protected-count / population
end

to log-setup
  show (word "Scenario " scenario-choice " | serotype " current-serotype " | Ro = " Ro " | VP = " VP)
  show (word "Population = " population " | density = " (precision density 2) " per km2 | contacts per animal = " n-contacts " | alpha = " alpha)
  show (word "Calibrated beta-daily = " (precision beta-daily 5) " | expected secondary infections = " (precision calibrated-R0 3) " (target " Ro ")")
  show (word "Carrier infectiousness = " carrier-infectiousness " | protection duration = " immunity-duration " d")
end

;; press CHECK after SETUP (or at any time) to run the built-in verification tests
to check-model
  update-compartment-counts
  update-outputs
  show (word "Population balance error = " balance-error "  (must be 0)")
  show (word "Calibrated expected secondary infections = " (precision calibrated-R0 3) "  (target Ro = " Ro ")")
  show (word "Doses = " doses-given " | protected (V responders) = " protected-count " | Peff = " (precision protected-fraction 4))
  show (word "Mean distance to contacts = " (precision (mean [ mean map [ nb -> distance nb ] nbr-list ] of turtles) 3) " km")
  show (word "Index animals = " (count turtles with [ initial-source? ]) "  (must be 1)")
end

;; ===== END COMMON ENGINE =====

;;##########################################################################
;; STRATEGY-SPECIFIC BLOCK: barrier
;; (procedures required by the engine: set-domain, place-animals, paint-domain,
;;  strategy-setup, strategy-daily)
;;##########################################################################

;; ---- domain: circular, r3 = 10 km, population from the density scenario ----
to set-domain
  set r3 10
  set area-km2 pi * (r3 ^ 2)
  if density-choice = "reference" [ set population 18536 ]
  if density-choice = "moderate"  [ set population 9268 ]
  if density-choice = "sparse"    [ set population 2001 ]
  set density population / area-km2
end

to place-animals
  ask turtles [
    let rr r3 * (sqrt (random-float 1))
    let th random-float 360
    setxy (rr * (sin th)) (rr * (cos th))
  ]
end

;; ---- barrier zone: Zone 1 = d < r1; Zone 2 (vaccinated) = r1 <= d <= r2 = r1 + w; Zone 3 = r2 < d <= r3 ----
to-report in-vaccination-zone?
  report ((distancexy 0 0) >= r1) and ((distancexy 0 0) <= (r1 + barrier-width))
end

to paint-domain
  ask patches [
    let d distancexy 0 0
    ifelse d > r3 [ set pcolor gray + 3 ] [ set pcolor white ]
    if d >= r1 and d <= (r1 + barrier-width) [ set pcolor green + 4.5 ]
  ]
end

;; ---- reactive campaign: day vaccination-day (after introduction), delivered over three days ----
;; On the first day campaign-coverage x (susceptible animals in the zone) are scheduled; one
;; third, then half of the remainder, then all of the remainder of the scheduled animals that
;; are still susceptible are vaccinated on days 0, 1 and 2 of the campaign.
;; Animals infected before their turn are not vaccinated (their dose is not used).
to strategy-setup
  set HIT 1 - (1 / Ro)
  set vaccination-coverage campaign-coverage
end

to strategy-daily
  if ticks = vaccination-day [
    let eligible turtles with [ state = "S" and not vaccinated? and in-vaccination-zone? ]
    ask n-of (round (vaccination-coverage * (count eligible))) eligible [ set scheduled? true ]
  ]
  if ticks >= vaccination-day and ticks < (vaccination-day + 3) [
    let share 1 / (3 - (ticks - vaccination-day))
    let pool turtles with [ scheduled? and not vaccinated? and state = "S" ]
    ask n-of (round (share * (count pool))) pool [ vaccinate ]
  ]
end

to-report zone-protected-fraction
  let z turtles with [ in-vaccination-zone? ]
  ifelse any? z [ report (count z with [ state = "V" ]) / (count z) ] [ report 0 ]
end
