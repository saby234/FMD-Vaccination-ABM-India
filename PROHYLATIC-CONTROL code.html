;;========================================
;; SPATIAL SCALE SETUP 
;;========================================
;; Computes meters-per-patch dynamically from the current NetLogo
;; world dimensions. Total area side length = 10 km (10000 m).
;; Animals are spaced 130 m apart. Infection radius is specified
;; in meters and converted to patch units, so the model behaves
;; consistently regardless of world size set in the interface.

to setup-spatial-scale
  set total-length-m 10000        ;; 10 km total
  set animal-spacing-m 130        ;; 130 m between adjacent animals
  set infection-radius-m 150      ;; infection radius in meters (slightly > spacing)

  ;; Derive meters-per-patch from actual world width.
  set meters-per-patch (total-length-m / world-width)

  ;; Convert meter-based quantities into patch units.
  set animal-spacing-patches (animal-spacing-m / meters-per-patch)
  set infection-radius (infection-radius-m / meters-per-patch)
end

to place-cows-in-grid
  let n population
  if n = 0 [ stop ]

  ;; spacing now driven by 130 m animal spacing (in patch units)
  let spacing animal-spacing-patches
  let rows floor (world-width / spacing)
  if rows < 1 [ set rows 1 ]
  let index 0

  ask turtles [
    let row index mod rows
    let col floor (index / rows)
    let x (row * spacing - world-width / 2)
    let y (col * spacing - world-height / 2)

    setxy x y

    set size 1.5
    set shape "cow"
    set state "S"
    set color brown
    set days-in-state 0
    set initial-source? false

    set index index + 1
  ]
end


to setup
  clear-all
  set-scenario

  set cumulative-fatalities 0
  set cumulative-recoveries 0

  ;; Compute meters-per-patch and meter-based radii dynamically.
  ;; This replaces the old hard-coded `set infection-radius 1`.
  setup-spatial-scale

  ;; Serotype parameters (durations + protectiveness)
  if current-serotype = "O" [
    set exposure-period 1.2
    set subclinical-period 1.7
    set clinical-period 6.6
    set VP 0.823
  ]
  if current-serotype = "A" [
    set exposure-period 1.0
    set subclinical-period 2.8
    set clinical-period 7.5
    set VP 0.767
  ]
  if current-serotype = "Asia1" [
    set exposure-period 1.5
    set subclinical-period 2.0
    set clinical-period 7.0
    set VP 0.787
  ]

  set carrier-duration 320
  set carrier-probability 0.5
  set case-fatality-rate 1.1e-5
  set immunity-duration 182
  set carrier-infectiousness 0.5

  ;; HIT and required coverage from Ro and VP
  compute-required-coverage

  ;; Step 1: Create cows
  create-turtles 5900 [
    set size 1.5
    set shape "cow"
    set state "S"
    set color brown
    set days-in-state 0
    set initial-source? false
  ]

  set population count turtles
  place-cows-in-grid

  ;; Step 2: Measure density BEFORE vaccination
  ;; (so we calibrate against true herd geometry, not post-vax)
  measure-avg-neighbors

  ;; Step 3: Calibrate beta-daily based on measured density
  let infectious-duration (subclinical-period + clinical-period)
  ifelse infectious-duration > 0 and avg-neighbors > 0 [
    set beta-daily Ro / (infectious-duration * avg-neighbors)
  ] [
    set beta-daily 0
  ]

  show (word "World: " world-width " x " world-height " patches | "
            "10 km total | "
            "meters-per-patch = " (precision meters-per-patch 3) " | "
            "spacing = " animal-spacing-m " m (" (precision animal-spacing-patches 3) " patches) | "
            "infection radius = " infection-radius-m " m (" (precision infection-radius 3) " patches)")
  show (word "avg-neighbors within " infection-radius-m " m: " precision avg-neighbors 3)
  show (word "calibrated beta-daily: " precision beta-daily 5)

  ;; Step 4: Vaccination (with VP)
  apply-vaccination

  ;; Step 5: Seed infection at center
  let candidate one-of turtles with [state = "S"]
  if candidate != nobody [
    ask candidate [
      setxy 0 0
      set state "I_clin"
      set color black
      set days-in-state 0
      set initial-source? true
    ]
  ]

  reset-ticks
end

;;========================================
;; MEASURE AVG NEIGHBORS WITHIN RADIUS
;;========================================
;; Sample 200 cows, count how many other cows are within
;; infection-radius. This estimates the local contact pool.

to measure-avg-neighbors
  let sample-size min list 200 (count turtles)
  let sample n-of sample-size turtles
  let total 0

  ask sample [
    let nearby other turtles in-radius infection-radius
    set total total + count nearby
  ]

  ifelse sample-size > 0 [
    set avg-neighbors total / sample-size
  ] [
    set avg-neighbors 0
  ]
end


;;========================================
;; SET-SCENARIO
;;========================================

to set-scenario
  ;; serotype and Ro only: HIT and the required coverage are COMPUTED in
  ;; compute-required-coverage (after VP is known), not typed in by hand
  if scenario-choice = "O_R2"     [ set current-serotype "O"     set Ro 2 ]
  if scenario-choice = "O_R4"     [ set current-serotype "O"     set Ro 4 ]
  if scenario-choice = "A_R2"     [ set current-serotype "A"     set Ro 2 ]
  if scenario-choice = "A_R4"     [ set current-serotype "A"     set Ro 4 ]
  if scenario-choice = "Asia1_R2" [ set current-serotype "Asia1" set Ro 2 ]
  if scenario-choice = "Asia1_R4" [ set current-serotype "Asia1" set Ro 4 ]
end

;;========================================
;; ARM 1 - CONTROL: NO VACCINATION
;;========================================
;; vaccination-coverage = 0: nobody is vaccinated and no animal is ever in state V.
;; HIT is computed for reference only.

to compute-required-coverage
  set HIT 1 - (1 / Ro)
  set vaccination-coverage 0
  show (word "CONTROL: Ro = " Ro " | HIT (reference only) = " (precision HIT 3)
            " | vaccination coverage = 0 (no vaccination)")
end

;;========================================
;; APPLY VACCINATION (VP-CORRECTED, IN-PLACE)
;;========================================

to apply-vaccination
  let num-vaccinated min list population round (vaccination-coverage * population)
  let vaccinated-cows n-of num-vaccinated turtles

  let success-count 0
  let failure-count 0

  ask vaccinated-cows [
    ifelse random-float 1 < VP [
      set state "V"
      set color green
      set days-in-state 0
      set success-count success-count + 1
    ] [
      set state "S"
      set color brown
      set days-in-state 0
      set failure-count failure-count + 1
    ]
  ]

  show (word "Vaccinated: " num-vaccinated
            " | Protected (V): " success-count
            " | Vaccine failures (still S): " failure-count
            " | VP = " VP)
  show (word "Realised protected fraction Peff = " (precision (success-count / population) 4)
            " | target HIT = " (precision HIT 3))
end

;;========================================
;; GO
;;========================================

to go
  calculate-rates

  if ticks >= 2 [
    update-turtles
  ]

  if ticks < 2 [
    ask turtles [
      set days-in-state days-in-state + 1
    ]
  ]

  update-compartment-counts
  tick
end

;;========================================
;; UPDATE TURTLES
;;========================================

to update-turtles

  ask turtles [
    set days-in-state days-in-state + 1
  ]

  ;; PASS 1: STATE PROGRESSION
  ask turtles [
    if state = "V" and days-in-state >= immunity-duration [
      set state "S"
      set color brown
      set days-in-state 0
    ]

    if state = "E" and days-in-state >= exposure-period [
      set state "I_sub"
      set color orange
      set days-in-state 0
    ]

    if state = "I_sub" and days-in-state >= subclinical-period [
      set state "I_clin"
      set color black
      set days-in-state 0
    ]

    if state = "I_clin" and days-in-state >= clinical-period [
      ifelse random-float 1 < carrier-probability [
        set state "C"
        set color gray
      ] [
        set state "R"
        set color cyan
      ]
      set days-in-state 0
    ]

    if state = "C" and days-in-state >= carrier-duration [
      set state "R"
      set color cyan
      set days-in-state 0
    ]
  ]

  ;; PASS 2: NEW INFECTIONS (RADIUS-BASED, CALIBRATED)
  ;; Each infectious cow exposes every S cow within
  ;; infection-radius with probability beta-daily per day.
  ;; Carriers contribute at reduced infectiousness.

  if ticks >= 2 [

    ;; Acute infectious (I_sub, I_clin): full beta
    ask turtles with [state = "I_sub" or state = "I_clin"] [
      let nearby-S other turtles in-radius infection-radius with [state = "S"]
      ask nearby-S [
        if random-float 1 < beta-daily [
          set state "E"
          set color yellow
          set days-in-state 0
        ]
      ]
    ]

    ;; Carriers: reduced beta (carrier-infectiousness multiplier)
    ask turtles with [state = "C"] [
      let nearby-S other turtles in-radius infection-radius with [state = "S"]
      ask nearby-S [
        if random-float 1 < (beta-daily * carrier-infectiousness) [
          set state "E"
          set color yellow
          set days-in-state 0
        ]
      ]
    ]
  ]
end

;;========================================
;; CALCULATE RATES
;;========================================

to calculate-rates
  let S count turtles with [state = "S"]
  let I-full count turtles with [state = "I_sub"] + count turtles with [state = "I_clin"]
  let I-carrier count turtles with [state = "C"]
  let N population
  if N = 0 [ set N 1 ]

  let effective-I I-full + (I-carrier * carrier-infectiousness)

  set Re_O Ro * S / N

  let infectious-duration (subclinical-period + clinical-period)
  ifelse infectious-duration > 0 [
    set transmission-rate Ro / infectious-duration
  ] [
    set transmission-rate 0
  ]

  set exposure-rate transmission-rate * effective-I / N

  ifelse exposure-period > 0 [
    set subclinical-infectious-rate (count turtles with [state = "E"]) / exposure-period
  ] [ set subclinical-infectious-rate 0 ]

  ifelse subclinical-period > 0 [
    set clinical-infectious-rate (count turtles with [state = "I_sub"]) / subclinical-period
  ] [ set clinical-infectious-rate 0 ]

  ifelse clinical-period > 0 [
    set carrier-rate (count turtles with [state = "I_clin"]) * (1 / clinical-period) * carrier-probability
    set recovery-rate (count turtles with [state = "I_clin"]) * (1 - carrier-probability) * (1 / clinical-period)
  ] [
    set carrier-rate 0
    set recovery-rate 0
  ]

  set fatality-rate case-fatality-rate * (count turtles with [state = "I_clin"])

  ifelse carrier-duration > 0 [
    set carrier-recovery-rate (count turtles with [state = "C"]) / carrier-duration
  ] [ set carrier-recovery-rate 0 ]

  set cumulative-fatalities cumulative-fatalities + fatality-rate
  set cumulative-recoveries cumulative-recoveries + recovery-rate + carrier-recovery-rate
end

;;========================================
;; UPDATE COMPARTMENT COUNTS
;;========================================

to update-compartment-counts
  set S-count      count turtles with [state = "S"]
  set E-count      count turtles with [state = "E"]
  set I-sub-count  count turtles with [state = "I_sub"]
  set I-clin-count count turtles with [state = "I_clin"]
  set C-count      count turtles with [state = "C"]
  set R-count      count turtles with [state = "R"]
  set V-count      count turtles with [state = "V"]
end
